# NVIDIA GPU install: local embeddings run on CUDA. The extras bring the CUDA and cuDNN DLLs (about 2 GB).
#
# Don't install this file directly. Stop the server, then run:
#   powershell -ExecutionPolicy Bypass -File scripts\install_gpu.ps1
#
# pip can't do this in one step: a CPU onnxruntime left in the venv shares a folder with
# onnxruntime-gpu and overwrites it, so embeddings would silently stay on the CPU. Afterwards, `pip check`
# Run the script again after any install that brings the CPU build back.
-r requirements-base.txt
fastembed-gpu
onnxruntime-gpu[cuda,cudnn]
pytest
pytest-cov
